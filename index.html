মেইন ফাংশন রান করার জন্য def main(): # তোমার বটের টোকেন এখানে বসিয়ে দিও TOKEN = "YOUR_BOT_TOKEN_HERE" app = ApplicationBuilder().token(TOKEN).build() app.add_handler(CommandHandler("start", start)) app.add_handler(CallbackQueryHandler(login_step, pattern="^do_login$")) app.add_handler(CallbackQueryHandler(tasks_menu, pattern="^menu_tasks$")) app.add_handler(CallbackQueryHandler(complete_task, pattern="^do_task_")) app.add_handler(CallbackQueryHandler(deposit_menu, pattern="^menu_deposit$")) app.add_handler(CallbackQueryHandler(confirm_deposit, pattern="^dep_")) app.add_handler(CallbackQueryHandler(profile_menu, pattern="^menu_profile$")) app.add_handler(CallbackQueryHandler(withdraw_menu, pattern="^menu_withdraw$")) app.add_handler(CallbackQueryHandler(manager_contact, pattern="^menu_manager$")) app.add_handler(CallbackQueryHandler(show_main_menu, pattern="^back_home$")) app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_user_input)) print("🤖 বট সফলভাবে চালু হয়েছে...") app.run_polling() if __name__ == "__main__": main()
